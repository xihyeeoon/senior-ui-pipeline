## 이전 시도의 실패

[고칠 것]
  원본의 pick-bank 선택지 67개 중 58개가 생성물에 없다 (예: BNK투자증권, BNP파리바, BOA, DB증권, HSBC …). 화면에 모두 보일 필요는 없지만 값 자체는 모두 접근 가능해야 한다. 검색이나 단계적 선택으로 찾을 수 있게 포함하라.

직전 출력을 고쳐라. 설계를 처음부터 새로 하지 마라.

function koreanAmount(n){
  n = Number(n)||0;
  if(!n) return '';
  const digit=['','일','이','삼','사','오','육','칠','팔','구'];
  const small=['','십','백','천'];
  const big=['','만','억'];
  let out='', g=0;
  while(n>0){
    const chunk=n%10000;
    if(chunk){
      let s='';
      if(!(chunk===1 && g>0)){
        const d=String(chunk).split('').reverse();
        for(let i=d.length-1;i>=0;i--){
          const v=+d[i];
          if(!v) continue;
          s += (v===1 && i>0 ? '' : digit[v]) + small[i];
        }
      }
      out = s + big[g] + ' ' + out;
    }
    n=Math.floor(n/10000); g++;
  }
  return out.trim() + ' 원';
}
