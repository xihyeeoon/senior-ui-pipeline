const LOG = [];

function show(name) {
  document.querySelectorAll('.screen').forEach(s => s.classList.toggle('on', s.dataset.screen === name));
  window.__task.screen = name;
  log('screen_enter', { to: name });
  window.scrollTo(0, 0);
}

window.__screen = () => window.__task.screen;

const S = { bank: '', account: '', amount: '', pw: '' };

function renderAccount() {
  document.getElementById('account-number').value = S.account;
}

function renderAmount() {
  document.getElementById('amount-input').value = Number(S.amount).toLocaleString();
}

function renderDots() { 
  document.querySelectorAll('#pw-dots i').forEach((d, i) => {
    d.className = i < S.pw.length ? 'fill' : (i === S.pw.length ? 'cur' : '');
  });
}

function buildPwPad() {
  const nums = [1, 2, 3, 4, 5, 6, 7, 8, 9, 0].sort(()=>Math.random()-0.5);
  const pad = document.getElementById('pw-pad');
  pad.innerHTML = nums.map(n => `<button data-action="pw" data-v="${n}">${n}</button>`).join('')
    + `<button data-action="pw-del">⌫</button>`;
}

document.getElementById('phone').addEventListener('click', function(e) {
  const el = e.target.closest('[data-action]');
  if (!el) return;
  const a = el.dataset.action;
  
  if (a === 'go-account') {
    S.bank = '';
    S.account = '';
    show('account');
  } else if (a === 'select-bank') {
    show('bank');
  } else if (a === 'pick-bank') {
    S.bank = el.dataset.bank;
    document.getElementById('bank-name').value = S.bank;
    show('account');
  } else if (a === 'acc-num') {
    if (S.account.length < 16) {
      S.account += el.dataset.v;
      renderAccount();
    }
  } else if (a === 'acc-del') {
    S.account = S.account.slice(0, -1);
    renderAccount();
  } else if (a === 'go-amount') {
    if(S.account && S.bank) {
      renderAmount();
      show('amount');
    }
  } else if (a === 'num') {
    if(S.amount.length < 9) {
      S.amount = (S.amount + el.dataset.v).replace(/^0+/, '');
      renderAmount();
    }
  } else if (a === 'del') {
    S.amount = S.amount.slice(0, -1);
    renderAmount();
  } else if (a === 'go-confirm') {
    if(S.amount) {
      document.getElementById('confirm-bank').textContent = S.bank;
      document.getElementById('confirm-account').textContent = S.account;
      document.getElementById('confirm-amount').textContent = Number(S.amount).toLocaleString();
      show('confirm');
    }
  } else if (a === 'go-password') {
    S.pw = '';
    buildPwPad();
    renderDots();
    show('password');
  } else if (a === 'pw') {
    if(S.pw.length < 4) {
      S.pw += el.dataset.v;

    log('task_end');
    show('start');
